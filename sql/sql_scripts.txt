-- Business Process 1: Flight Reservation & Revenue
CREATE TABLE FLIGHT_RESERVATION_FACT (
    fact_key NUMBER PRIMARY KEY,
    date_key NUMBER(8) NOT NULL,
    passenger_key NUMBER NOT NULL,
    flight_key NUMBER NOT NULL,
    route_key NUMBER NOT NULL,
    aircraft_key NUMBER NOT NULL,
    fare_class_key NUMBER NOT NULL,
    channel_key NUMBER NOT NULL,
    promo_key NUMBER NOT NULL, -- Defaults to -1 in ETL if no promotion was used
    
    -- Measures
    ticket_revenue NUMBER(15,2) NOT NULL,
    flight_cost NUMBER(15,2) NOT NULL,
    flight_profit NUMBER(15,2) NOT NULL,
    miles_earned NUMBER NOT NULL,
    miles_redeemed NUMBER NOT NULL,
    delay_minutes NUMBER NOT NULL,
    is_delayed_flag NUMBER(1) NOT NULL,
    
    -- Foreign Key Constraints
    CONSTRAINT fk_res_date FOREIGN KEY (date_key) REFERENCES DATE_DIM(date_key),
    CONSTRAINT fk_res_pass FOREIGN KEY (passenger_key) REFERENCES PASSENGER_DIM(passenger_key),
    CONSTRAINT fk_res_flight FOREIGN KEY (flight_key) REFERENCES FLIGHT_DIM(flight_key),
    CONSTRAINT fk_res_route FOREIGN KEY (route_key) REFERENCES ROUTE_AIRPORT_DIM(route_key),
    CONSTRAINT fk_res_air FOREIGN KEY (aircraft_key) REFERENCES AIRCRAFT_DIM(aircraft_key),
    CONSTRAINT fk_res_fare FOREIGN KEY (fare_class_key) REFERENCES FARE_CLASS_DIM(fare_class_key),
    CONSTRAINT fk_res_chan FOREIGN KEY (channel_key) REFERENCES BOOKING_CHANNEL_DIM(channel_key),
    CONSTRAINT fk_res_promo FOREIGN KEY (promo_key) REFERENCES PROMOTION_DIM(promo_key)
);

INSERT INTO "DWH"."DATE_DIM"
SELECT 
    TO_NUMBER(TO_CHAR(curr_date, 'YYYYMMDD')) AS "DATE_KEY",
    curr_date AS "BUS_DATE",
    curr_date AS "FULLDATE",
    EXTRACT(YEAR FROM curr_date) AS "YEAR",
    TO_NUMBER(TO_CHAR(curr_date, 'Q')) AS "QUARTER",
    TO_CHAR(curr_date, 'Month', 'NLS_DATE_LANGUAGE=ENGLISH') AS "MONTHNAME",
    TO_CHAR(curr_date, 'Day', 'NLS_DATE_LANGUAGE=ENGLISH') AS "DAYOFWEEK",
    'N/A' AS "SSC" -- Replace 'N/A' with your required system/season code logic if needed
FROM (
    SELECT TO_DATE('2015-01-01', 'YYYY-MM-DD') + LEVEL - 1 AS curr_date
    FROM DUAL
    CONNECT BY LEVEL <= (TO_DATE('2035-12-31', 'YYYY-MM-DD') - TO_DATE('2015-01-01', 'YYYY-MM-DD') + 1)
);

COMMIT;

-- 1. Create the sequence
CREATE SEQUENCE DWH.SEQ_PASSENGER_KEY 
START WITH 1 
INCREMENT BY 1;

-- 2. Create the before-insert trigger
CREATE OR REPLACE TRIGGER DWH.TRG_PASSENGERS_DIM_BI
BEFORE INSERT ON DWH.PASSENGERS_DIM
FOR EACH ROW
BEGIN
  IF :NEW.PASSENGER_KEY IS NULL OR :NEW.PASSENGER_KEY = 0 THEN
    :NEW.PASSENGER_KEY := DWH.SEQ_PASSENGER_KEY.NEXTVAL;
  END IF;
END;
/
